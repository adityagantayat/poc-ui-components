To use the Toast notifications in your component, import from the appropriate path

```tsx
import ToastManager from './toast-notifications/ToastContainer';
```
 A very simple and straightforward implementation
```tsx
const MyComponent: React.FC = () => {
    const toastManagerRef = useRef<any>(null);
    const handleAddToast = (variant: 'success' | 'error' | 'warning' | 'info') => 
    {
      if (toastManagerRef.current) {
        //The addToast function accepts
        // message: string, 
        // variant:'success' | 'error' | 'warning' | 'info',
        // animation: 'slide' | 'fade' | 'bounce' | 'pop'
        // mode: 'dark' | 'light'
        // icon: React.ReactNode --(any Icon that you would like to be displayed before the toast message)
        toastManagerRef.current.addToast("This is a toast message", 'success', 'slide', 'dark', 
                                        <CircleCheckBig color='green'/> );
      }
    };
    return (
        <div>
            <button className="bg-blue-500 text-white px-4 py-2 rounded"
                onClick={() => {
                    handleAddToast('success');
                }}
            >
                Show Toast
            </button>
            {/* include the ToastManager anywhere in your component and add the 
            maximum number of toasts you would like to see on your screen at once */}
            <ToastManager ref={toastManagerRef} maxToasts={3}/>
        </div>
    );
};
```
